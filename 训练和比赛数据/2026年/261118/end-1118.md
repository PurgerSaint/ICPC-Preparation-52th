bool solve() {
    int n; cin >> n;
    vector<pair<ll, ll>> b(n);
    for (auto& [a, m]: b) cin >> a >> m;
    sort(range(b), [](pair<ll, ll>& x, pair<ll, ll>& y){
        if (x.second == y.second) return x.first > y.first;
        return x.second > y.second;
    });
    auto& [am, mx] = b[0];

    for (auto it = b.begin() + 1; it != b.end(); it++) {

}
