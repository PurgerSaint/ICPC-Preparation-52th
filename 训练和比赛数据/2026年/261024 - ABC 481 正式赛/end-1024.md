# F.Evil Shortest Path
## 题目详情
You are given a directed graph with $N$ vertices and $M$ edges. The vertices are numbered $1,2,\ldots,N$. The $i$\-th edge goes from vertex $A_i$ to vertex $B_i$, and a digit $C_i$ between $1$ and $9$, inclusive, is written on this edge.
When moving along the edges in their directions, the integer obtained by concatenating the digits written on the traversed edges in the order they are traversed is called the **score** of the movement. For example, if the digits written on the traversed edges are $3,1,4$ in this order, the **score** of the movement is $314$. The same vertex or edge may be traversed multiple times.
For each $v=2,3,\ldots,N$, find the minimum **score** of a movement from vertex $1$ to vertex $v$, modulo $998244353$. Here, it is guaranteed that the given directed graph allows moving from vertex $1$ to every other vertex along the edges in their directions.
*Constraints*
-   $2 \le N \le 2\times 10^5$
-   $N-1 \le M \le 2\times 10^5$
-   $1 \le A_i,B_i \le N$
-   $1 \le C_i \le 9$
-   All input values are integers.
-   It is possible to move from vertex $1$ to every other vertex along the edges in their directions.

```cpp
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
#define ios ios::sync_with_stdio(false);cin.tie(0);
#define mod(t, m) ((t) % (m) + (m)) % (m)
#define range(a) a.begin(), a.end()

struct state {
    int v; ll w;
    int cnt;
    bool operator<(const state& other) const {
        if (cnt != other.cnt) return cnt > other.cnt;
        return w > other.w;
    }
};

int main() {
    ios;
    int n, m; cin >> n >> m;
    vector<vector<pair<int, int>>> mp(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, c; cin >> u >> v >> c;
        mp[u].push_back({v, c});
    }
    const ll INF = 1LL << 60;
    const ll MOD = 998244353;
    vector<ll> dis(n + 1, INF);
    vector<ll> cnt(n + 1, INF);
    priority_queue<state> q;
    q.push({1, 0, 0});
    dis[1] = 0;
    cnt[1] = 0;
    while (!q.empty()) {
        state t = q.top(); q.pop();
        auto& [u, w, ct] = t;
        if (w != dis[u] || ct != cnt[u]) continue;
        for (auto& [v, c]: mp[u]) {
            ll w2 = w * 10 + c;
            int n_ct = ct;
            if (w2 >= MOD) n_ct += w2 / MOD;
            w2 = mod(w2, MOD);

            if (n_ct < cnt[v] || (n_ct == cnt[v] && w2 < dis[v])) {
                dis[v] = w2;
                cnt[v] = n_ct;
                q.push({v, dis[v], n_ct});
            }
        }
    }
    for (int i = 2; i <= n; i++)
        cout << dis[i] << (i < n ? "\n" : "");
    return 0;
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
using ll = long long;
#define ios ios::sync_with_stdio(false);cin.tie(0);
#define mod(t, m) ((t) % (m) + (m)) % (m)
#define range(a) a.begin(), a.end()

int main() {
    ios;
    int n, m; cin >> n >> m;
    vector<vector<pair<int, int>>> mp(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, c; cin >> u >> v >> c;
        mp[u].push_back({v, c});
    }

    const ll INF = 1LL << 60;
    const ll MOD = 998244353;
    
    vector<ll> dis(n + 1, INF);
    vector<bool> vis(n + 1, 0);

    queue<vector<int>> q;
    q.push({1});
    vis[1] = 1;
    dis[1] = 0;

    while (!q.empty()) {
        vector<int> V = q.front(); q.pop();

        vector<vector<int>> next_node(10);
        for (int& u: V) {
            for (auto& [v, c]: mp[u]) {
                if (!vis[v])
                    next_node[c].push_back(v);
            }
        }

        for (int c = 1; c <= 9; c++) {
            if (next_node[c].empty()) continue;

            vector<int> next_V;
            for (int& v: next_node[c]) {
                if (vis[v]) continue;
                vis[v] = 1;
                dis[v] = mod(dis[V[0]] * 10LL + c, MOD);
                next_V.push_back(v);
            }
            if (!next_V.empty()) q.push(next_V);
        }
    }
    for (int i = 2; i <= n; i++)
        cout << dis[i] << (i < n ? "\n" : "");
    return 0;
}
```
